ENV DEBIAN_FRONTEND=noninteractive

    libc6=2.31-0ubuntu9.5 \

    zlib1g=1:1.2.12.dfsg-8ubuntu1.20 \

RUN pip install --no-cache-dir Flask==2.2.4 PyYAML==3.13
WORKDIR /srv/app
COPY . .
EXPOSE 8080
CMD ["./entrypoint.sh"]
